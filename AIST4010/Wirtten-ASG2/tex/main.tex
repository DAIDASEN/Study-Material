\section{Question 1}
\textbf{Definitions.}
\begin{itemize}[leftmargin=*]
    \item \emph{Vanishing gradients}: Backpropagated gradients decay exponentially across layers (often with saturating activations), yielding negligible updates in early layers.
    \item \emph{Exploding gradients}: Backpropagated gradients grow exponentially, causing unstable updates and divergence.
\end{itemize}

\textbf{Mitigations (at least three).}
\begin{itemize}[leftmargin=*]
    \item \emph{Proper initialization}: Xavier/Glorot for tanh; He/Kaiming for ReLU-family to preserve variance through layers.
    \item \emph{Activation choice}: Prefer non-saturating activations (ReLU, LeakyReLU, GELU); avoid deep stacks of sigmoid/tanh.
    \item \emph{Normalization}: BatchNorm/LayerNorm stabilize activations and gradients, enabling deeper networks and larger learning rates.
    \item \emph{Gradient clipping}: Clip global norm or per-parameter norms to cap extreme updates (mainly for explosion).
    \item \emph{Architectural aids}: Residual/skip connections and gating (e.g., LSTM) shorten gradient paths and mitigate decay.
\end{itemize}


\section{Question 2}
\textbf{Similarities.}
\begin{itemize}[leftmargin=*]
  \item \emph{Locality and parameter sharing}: Convolutional kernels apply local receptive fields with shared weights.
  \item \emph{Translation equivariance}: Feature maps shift consistently with input translations (exact under appropriate padding/stride settings).
\end{itemize}

\textbf{Differences.}
\begin{itemize}[leftmargin=*]
  \item \emph{Data domain}: 2D conv on H$\times$W (e.g., images); 3D conv on H$\times$W$\times$D/T (e.g., volumes/videos).
  \item \emph{Kernel sweep}: 2D slides over two axes; 3D slides over three axes, capturing volumetric or spatio-temporal patterns.
  \item \emph{Compute/params}: 3D kernels incur higher cost and memory; risk of overfitting increases under limited data.
  \item \emph{Use cases}: 2D for spatial cues (edges, textures); 3D for motion/shape continuity (temporal dynamics, 3D anatomy).
  \item \emph{Design variants}: Factorized (2+1)D or separable spatio-temporal convolutions reduce cost while preserving accuracy in videos.
\end{itemize}


\section{Question 3}
\textbf{Symptoms.} Loss plateaus or oscillates in early epochs without downward trend.

\textbf{Likely causes (examples).}
\begin{itemize}[leftmargin=*]
  \item \emph{Learning rate mis-specified}: too low (minuscule updates) or too high (overshooting/oscillation). Consider LR range test or scheduler warmup.
  \item \emph{Data/label issues}: missing normalization/standardization; severe label noise or class leakage; erroneous preprocessing/augmentations.
  \item \emph{Initialization/activation mismatch}: poor initialization (symmetry, saturation), incompatible with activation (e.g., sigmoid + large weights).
  \item \emph{Optimization setup}: wrong loss (e.g., missing reduction/weights), disabled gradient flow (requires\_grad/ detach), weight decay too large; model accidentally in eval mode (BatchNorm/Dropout frozen).
  \item \emph{Batching/runtime}: batch size too small/large; exploding gradients with silent NaNs (check gradient norms, enable anomaly detection).
\end{itemize}

\section{Question 4}
\textbf{Effects.}
\begin{itemize}[leftmargin=*]
    \item \emph{Biased classifiers}: Decision boundaries skew toward majority class; accuracy may be misleading.
    \item \emph{Poor minority recall/precision}: Elevated FN/FP for rare classes; unstable calibration.
\end{itemize}

\textbf{Alleviation (at least three).}
\begin{itemize}[leftmargin=*]
    \item \emph{Resampling}: Minority oversampling (e.g., SMOTE/ADASYN) or majority undersampling; combine with stratified batching.
    \item \emph{Cost-sensitive learning}: Class-weighted losses (e.g., weighted cross-entropy, focal loss) to penalize minority errors more.
    \item \emph{Data augmentation}: Targeted augmentations for minority classes to increase diversity and reduce overfitting.
    \item \emph{Thresholding/metrics}: Tune decision thresholds per class; report AUROC, AUPRC, F1 instead of accuracy.
    \item \emph{Ensembles}: Bagging/boosting with balanced sampling to improve minority robustness.
\end{itemize}
