\textbf{Effects.}
\begin{itemize}[leftmargin=*]
    \item \emph{Biased classifiers}: Decision boundaries skew toward majority class; accuracy may be misleading.
    \item \emph{Poor minority recall/precision}: Elevated FN/FP for rare classes; unstable calibration.
\end{itemize}

\textbf{Alleviation (at least three).}
\begin{itemize}[leftmargin=*]
    \item \emph{Resampling}: Minority oversampling (e.g., SMOTE/ADASYN) or majority undersampling; combine with stratified batching.
    \item \emph{Cost-sensitive learning}: Class-weighted losses (e.g., weighted cross-entropy, focal loss) to penalize minority errors more.
    \item \emph{Data augmentation}: Targeted augmentations for minority classes to increase diversity and reduce overfitting.
    \item \emph{Thresholding/metrics}: Tune decision thresholds per class; report AUROC, AUPRC, F1 instead of accuracy.
    \item \emph{Ensembles}: Bagging/boosting with balanced sampling to improve minority robustness.
\end{itemize}
