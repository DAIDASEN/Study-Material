\textbf{Efficient Estimation of Word Representations in Vector Space (Word2Vec).}
\begin{itemize}[leftmargin=*]
    \item \emph{Motivation}: Scalable training of high-quality word embeddings on web-scale corpora.
    \item \emph{Novelty}: CBOW and Skip-gram objectives; efficient approximations (hierarchical softmax, negative sampling).
    \item \emph{Intuition}: Distributional hypothesis—context prediction yields semantic/syntactic regularities in embedding space.
    \item \emph{Implementation}: \href{https://code.google.com/archive/p/word2vec/}{official archive}.
\end{itemize}

\textbf{Attention Is All You Need (Transformer).}
\begin{itemize}[leftmargin=*]
    \item \emph{Motivation}: Remove recurrence/convolution to improve parallelism and long-range dependency modeling in translation.
    \item \emph{Novelty}: Pure self-attention encoder–decoder, multi-head attention, positional encoding, scaled dot-product.
    \item \emph{Intuition}: Each token attends to all others, capturing global dependencies with parallel computation.
    \item \emph{Implementation}: \href{https://github.com/tensorflow/tensor2tensor}{Tensor2Tensor}.
\end{itemize}

\textbf{BERT: Pre-training of Deep Bidirectional Transformers.}
\begin{itemize}[leftmargin=*]
    \item \emph{Motivation}: Learn deeply bidirectional language representations adaptable to diverse NLU tasks.
    \item \emph{Novelty}: Masked Language Modeling and Next Sentence Prediction pretraining objectives.
    \item \emph{Intuition}: Conditioning on both left and right context yields richer token representations transferable to downstream tasks.
    \item \emph{Implementation}: \href{https://github.com/google-research/bert}{google-research/bert}.
\end{itemize}

\textbf{MSA Transformer.}
\begin{itemize}[leftmargin=*]
    \item \emph{Motivation}: Exploit multiple sequence alignments to predict protein structural signals (e.g., contacts).
    \item \emph{Novelty}: Row/column attention across sequences and aligned positions; captures co-evolutionary patterns.
    \item \emph{Intuition}: Correlated mutations imply spatial proximity; attention over MSAs models these dependencies.
    \item \emph{Implementation}: \href{https://github.com/facebookresearch/esm}{facebookresearch/esm} (ESM suite).
\end{itemize}
